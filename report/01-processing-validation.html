resolve_pooled_path <- function() {
  candidates <- c(
    "proc/output/pooled_data.rds",
    "output/pooled_data.rds",
    "../proc/output/pooled_data.rds",
    "../../proc/output/pooled_data.rds"
  )

  hits <- candidates[file.exists(candidates)]
  if (length(hits) == 0) {
    stop(
      "pooled_data.rds not found. Tried: ",
      paste(candidates, collapse = ", ")
    )
  }

  normalizePath(hits[1], winslash = "/", mustWork = TRUE)
}

render_kable_scroll <- function(df, caption, digits = NULL, max_height = "420px") {
  tbl <- knitr::kable(
    as.data.frame(df),
    format = "html",
    caption = caption,
    digits = digits,
    row.names = FALSE
  )

  if (requireNamespace("kableExtra", quietly = TRUE)) {
    tbl <- kableExtra::kable_styling(
      tbl,
      full_width = FALSE,
      bootstrap_options = c("striped", "hover", "condensed", "responsive"),
      fixed_thead = TRUE
    )
    tbl <- kableExtra::scroll_box(tbl, width = "100%", height = max_height)
  } else {
    tbl <- htmltools::div(
      style = paste0(
        "overflow:auto; max-height:", max_height,
        "; border:1px solid #d9d9d9; padding:6px;"
      ),
      htmltools::HTML(as.character(tbl))
    )
  }

  tbl
}

id_vars <- c("idstudent", "idschool", "idcountry")

is_qualitative <- function(x, nm) {
  if (nm %in% id_vars) return(FALSE)
  if (is.character(x) || is.factor(x) || is.logical(x)) return(TRUE)
  if (is.numeric(x) || is.integer(x)) {
    ux <- unique(x[!is.na(x)])
    return(length(ux) <= 12)
  }
  FALSE
}

qual_vars <- names(db)[map2_lgl(db, names(db), is_qualitative)]

freq_table <- function(data, var) {
  data |>
    mutate(value = as.character(.data[[var]])) |>
    mutate(value = if_else(is.na(value), "NA", value)) |>
    count(value, name = "n", sort = TRUE) |>
    mutate(
      variable = var,
      pct = round(100 * n / sum(n), 2)
    ) |>
    select(variable, value, n, pct)
}

is_quantitative <- function(x, nm) {
  if (nm %in% id_vars) return(FALSE)
  if (!(is.numeric(x) || is.integer(x))) return(FALSE)
  ux <- unique(x[!is.na(x)])
  length(ux) > 12
}

quant_vars <- names(db)[map2_lgl(db, names(db), is_quantitative)]

safe_skewness <- function(x) {
  x <- x[!is.na(x)]
  n <- length(x)
  if (n < 3) return(NA_real_)
  s <- sd(x)
  if (is.na(s) || s == 0) return(NA_real_)
  m <- mean(x)
  mean(((x - m) / s)^3)
}

safe_kurtosis <- function(x) {
  x <- x[!is.na(x)]
  n <- length(x)
  if (n < 4) return(NA_real_)
  s <- sd(x)
  if (is.na(s) || s == 0) return(NA_real_)
  m <- mean(x)
  mean(((x - m) / s)^4) - 3
}

if (length(quant_vars) == 0) {

    ) |>
